## Tests first

Run the capture before any edit. Save the block below as `/tmp/s1912-check.sh`,
`chmod +x` it, and run `/tmp/s1912-check.sh before`; run
`/tmp/s1912-check.sh after` after the edits. It compares headings (the only
allowed removals are the repeated `Run it`, `Build`, `Usage`, `Schema` and
`Steps` subheadings), `**State**` lines, every fenced block inside an Acceptance
section plus every Gherkin block (byte for byte), and the set of `S`, `F` and
`E` IDs each page mentions, and it checks that every anchor link from `docs/`
into the six pages still names a heading.

````bash
#!/usr/bin/env bash
# usage: chk.sh before   (run once, before any edit)
#        chk.sh after    (run after the edits)
set -euo pipefail
E=docs/specs/epics
PAGES="$E/E17-kiln-adoption/index.md $E/E17-kiln-adoption/F17.1-docs-model.md $E/E17-kiln-adoption/F17.2-cutover.md $E/E17-kiln-adoption/F17.5-consumer-checks-in-kiln.md $E/E11-workspace-generators/F11.1-ng-plugin.md $E/E12-on-demand-features/index.md"
B=/tmp/s1912-before
A=/tmp/s1912-after
ALLOWED='^#{3,4} (Run it|Build|Usage|Schema|Steps)$'

snap() {
  rm -rf "$1"; mkdir -p "$1"
  for p in $PAGES; do
    n=$(basename "$p")_$(basename "$(dirname "$p")")
    grep -E '^#{1,6} ' "$p" | sort > "$1/$n.headings"
    { grep -E '^\*\*State' "$p" || true; } > "$1/$n.states"
    { grep -oE '\b(S[0-9]+\.[0-9]+\.[0-9]+|F[0-9]+\.[0-9]+|E[0-9]+)\b' "$p" || true; } | sort -u > "$1/$n.ids"
    awk '
      /^(\*\*Acceptance|#{2,3} Acceptance)/ {acc=1}
      /^#{2,3} / && !/Acceptance/ {acc=0}
      /^\*\*(Build|Steps|Fallbacks|Known risk)/ {acc=0}
      /^```/ { if (!inb) {inb=1; keep=(acc || $0 ~ /gherkin/)} else {if (keep) print; inb=0; keep=0; next} }
      inb && keep {print}
    ' "$p" > "$1/$n.accblocks"
  done
}

if [ "${1:-}" = before ]; then snap "$B"; echo "captured $B"; exit 0; fi

snap "$A"
fail=0
for f in "$B"/*.headings; do
  n=$(basename "$f")
  [ -z "$(comm -23 <(grep -vE "$ALLOWED" "$f" | sort) "$A/$n")" ] || { echo "HEADING LOST in $n:"; comm -23 <(grep -vE "$ALLOWED" "$f" | sort) "$A/$n"; fail=1; }
  [ -z "$(comm -13 "$f" "$A/$n")" ] || { echo "HEADING ADDED/RENAMED in $n:"; comm -13 "$f" "$A/$n"; fail=1; }
done
for k in states accblocks; do
  for f in "$B"/*."$k"; do n=$(basename "$f"); cmp -s "$f" "$A/$n" || { echo "$k CHANGED: $n"; fail=1; }; done
done
for f in "$B"/*.ids; do
  n=$(basename "$f"); m=$(comm -23 "$f" "$A/$n" | tr '\n' ' ')
  [ -z "$m" ] || { echo "ID no longer mentioned in $n: $m"; fail=1; }
done

# every anchor link into these pages, from anywhere under docs/ or the root docs, still names a heading
python3 -I - $PAGES <<'PY' || fail=1
import pathlib, re, sys
targets = {pathlib.Path(p).resolve(): p for p in sys.argv[1:]}
def slugs(text):
    seen, out = {}, set()
    for line in text.splitlines():
        m = re.match(r'#{1,6}\s+(.*?)\s*#*\s*$', line)
        if not m: continue
        t = re.sub(r'\[([^\]]*)\]\([^)]*\)', r'\1', m.group(1)).replace('`', '')
        s = re.sub(r'[-\s]+', '-', re.sub(r'[^\w\s-]', '', t).strip().lower())
        k = seen.get(s, 0); seen[s] = k + 1
        out.add(s if k == 0 else f'{s}_{k}')
    return out
heads = {p: slugs(p.read_text()) for p in targets}
bad = 0
for md in pathlib.Path('docs').rglob('*.md'):
    for m in re.finditer(r'\]\(([^)#\s]*)#([^)\s]+)\)', md.read_text()):
        path, frag = m.groups()
        dest = md.resolve() if path == '' else (md.parent / path).resolve()
        if dest in targets and frag not in heads[dest]:
            print(f'BROKEN ANCHOR {md}: {path}#{frag}'); bad = 1
sys.exit(bad)
PY
exit $fail
````
